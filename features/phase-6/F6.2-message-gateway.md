# F6.2 — 消息平台网关

> **Phase**: Phase 6 — 定时工作流 + 消息网关 (Workflow & Gateway)
> **Priority**: P0
> **Status**: Not Started
> **Dependencies**: F6.1 (定时工作流引擎), F4.1 (多层级记忆系统)

## 概述

消息平台网关是 Agent Buddy 的核心差异化能力之一——让 Agent 的能力从桌面端延伸到消息平台（微信/QQ/钉钉/飞书/企业微信）。用户可以在微信里发消息调起 Agent 执行任务，Agent 的结果直接推送到消息平台。网关包含渠道适配器（各平台协议封装）、消息路由（外部消息→Agent 映射）、会话映射（外部 ID↔内部会话）、凭据管理（Bot Token 加密存储）和消息格式转换（各平台格式↔统一内部格式）。

关键设计包括跨渠道记忆（记忆标记来源渠道，桌面端打开能看到完整对话历史）、DM 安全策略（pairing/open/closed 三种模式，未知发送人需配对确认）、流式响应推送（支持逐字推送的平台实时推送）和工具权限降级（通过消息渠道触发的会话工具权限自动降级）。这让 Agent Buddy 成为一个真正的跨平台 AI 工作伙伴。

```typescript
// 渠道配置
interface ChannelConfig {
  id: string;
  platform: "dingtalk" | "qq" | "wecom" | "feishu" | "wechat" | "webhook";
  name: string;
  credentials: {
    // 各平台不同的凭据字段
    appId?: string;
    appSecret?: string;
    token?: string;
    encodingAESKey?: string;
    botId?: string;
  };

  // 路由规则
  routing: {
    // 默认处理 Agent
    defaultAgentId: string;
    // 按来源路由（可选）
    rules?: RoutingRule[];
  };

  // 消息处理
  messageHandling: {
    // 群聊是否只响应 @机器人
    groupMentionOnly: boolean;
    // 是否接受图片/文件
    acceptMedia: boolean;
    // 流式响应（逐字推送）
    streaming: boolean;
  };

  // DM 安全策略
  dmPolicy: "pairing" | "open" | "closed";

  enabled: boolean;
  status: "connected" | "disconnected" | "error";
}

// 路由规则
interface RoutingRule {
  condition: {
    type: "keyword" | "sender" | "group";
    value: string;
  };
  agentId: string;
}

// 渠道-会话映射
interface ChannelSessionMapping {
  id: string;
  channelConfigId: string;
  externalUserId: string;    // 外部平台的用户 ID
  externalGroupId?: string;  // 外部平台的群 ID
  sessionId: string;         // 内部会话 ID
  paired: boolean;           // 是否已配对确认
  createdAt: number;
}

// 统一内部消息格式
interface UnifiedMessage {
  id: string;
  channelConfigId: string;
  externalUserId: string;
  externalGroupId?: string;
  externalMessageId?: string;  // 外部平台消息 ID（用于回复）
  content: string;               // 文本内容
  media?: {
    type: "image" | "file" | "audio";
    url: string;
    name?: string;
  }[];
  isGroupMessage: boolean;
  mentionedBot: boolean;
  timestamp: number;
}

// 适配器接口（所有平台适配器实现此接口）
interface ChannelAdapter {
  platform: string;
  connect(config: ChannelConfig): Promise<void>;
  disconnect(): Promise<void>;
  sendMessage(target: { userId?: string; groupId?: string }, message: string, opts?: { streaming?: boolean }): Promise<void>;
  onMessage(callback: (message: UnifiedMessage) => void): void;
  getStatus(): "connected" | "disconnected" | "error";
}
```

### 关键实现逻辑

**Gateway 网关核心架构**：

```typescript
// main/services/gateway/MessageGateway.ts

class MessageGateway {
  private adapters: Map<string, ChannelAdapter> = new Map();
  private mappings: ChannelSessionMapping[] = [];

  // 初始化所有启用的渠道
  async initialize(): Promise<void> {
    const channels = await this.db.getAllChannels();
    for (const channel of channels.filter(c => c.enabled)) {
      await this.connectChannel(channel);
    }
  }

  // 连接渠道
  async connectChannel(config: ChannelConfig): Promise<void> {
    const adapter = this.createAdapter(config.platform);
    await adapter.connect(config);

    // 注册消息回调
    adapter.onMessage((message) => this.handleIncomingMessage(message));

    this.adapters.set(config.id, adapter);
  }

  // 处理收到的消息
  private async handleIncomingMessage(message: UnifiedMessage): Promise<void> {
    const channel = await this.db.getChannel(message.channelConfigId);

    // 1. 安全检查
    if (message.isGroupMessage && channel.messageHandling.groupMentionOnly && !message.mentionedBot) {
      return; // 群聊且未 @机器人，忽略
    }

    // 2. DM 安全策略检查
    if (!message.isGroupMessage) {
      if (channel.dmPolicy === "closed") return;

      let mapping = await this.findMapping(message.channelConfigId, message.externalUserId);
      if (!mapping) {
        if (channel.dmPolicy === "pairing") {
          // 创建待配对映射，发送配对码
          mapping = await this.createPendingMapping(message);
          await this.sendPairingCode(message, mapping);
          return;
        } else if (channel.dmPolicy === "open") {
          mapping = await this.createMapping(message);
        }
      } else if (!mapping.paired) {
        // 已有映射但未配对，忽略
        return;
      }
    }

    // 3. 路由到 Agent
    const agentId = this.routeToAgent(channel, message);
    const sessionId = await this.getOrCreateSession(message);

    // 4. 创建会话或复用已有会话
    const session = await this.getOrCreateSession(message, agentId, channel);

    // 5. 注入跨渠道记忆
    const memoryContext = await memoryManager.buildMemoryContext(session.projectId);

    // 6. 执行 Agent（工具权限降级）
    const result = await agentManager.prompt(session.id, message.content, {
      toolPermissionLevel: "restricted", // 权限降级
      sourceChannel: channel.platform,
      memoryContext,
    });

    // 7. 格式转换并发送回复
    const adapter = this.adapters.get(message.channelConfigId)!;
    if (channel.messageHandling.streaming && result.streaming) {
      // 流式推送
      await this.streamResponse(adapter, message, result.stream);
    } else {
      await adapter.sendMessage(
        { userId: message.externalUserId, groupId: message.externalGroupId },
        result.text
      );
    }
  }

  // 路由到 Agent
  private routeToAgent(channel: ChannelConfig, message: UnifiedMessage): string {
    if (channel.routing.rules) {
      for (const rule of channel.routing.rules) {
        if (this.matchesRule(rule, message)) {
          return rule.agentId;
        }
      }
    }
    return channel.routing.defaultAgentId;
  }

  // 获取或创建会话映射
  private async getOrCreateSession(message: UnifiedMessage, agentId: string, channel: ChannelConfig): Promise<Session> {
    let mapping = await this.findMapping(message.channelConfigId, message.externalUserId);

    if (mapping) {
      // 复用已有会话
      return await sessionManager.getSession(mapping.sessionId);
    }

    // 创建新会话
    const session = await sessionManager.createSession({
      agentId,
      title: `[${channel.platform}] ${message.externalUserId}`,
      sourceChannel: channel.platform,
    });

    // 创建映射
    await this.createMapping(message, session.id);

    return session;
  }
}
```

**钉钉 Stream 模式适配器**：

```typescript
// main/services/gateway/adapters/DingTalkAdapter.ts

import { DingTalkStream } from "dingtalk-stream";

class DingTalkAdapter implements ChannelAdapter {
  platform = "dingtalk";
  private client: DingTalkStream | null = null;
  private messageCallback: ((message: UnifiedMessage) => void) | null = null;

  async connect(config: ChannelConfig): Promise<void> {
    this.client = new DingTalkStream({
      clientId: config.credentials.appId!,
      clientSecret: config.credentials.appSecret!,
    });

    // 注册消息回调
    this.client.registerCallback((event) => {
      const message: UnifiedMessage = {
        id: generateId(),
        channelConfigId: config.id,
        externalUserId: event.senderId,
        externalGroupId: event.conversationId,
        externalMessageId: event.msgId,
        content: event.text?.content || "",
        isGroupMessage: event.conversationType === "2",
        mentionedBot: this.isMentioned(event),
        timestamp: Date.now(),
      };
      this.messageCallback?.(message);
    });

    await this.client.connect();
  }

  async sendMessage(target: { userId?: string; groupId?: string }, message: string): Promise<void> {
    // 调用钉钉 API 发送消息
    if (target.groupId) {
      await this.sendGroupMessage(target.groupId, message);
    } else if (target.userId) {
      await this.sendDirectMessage(target.userId, message);
    }
  }

  async sendStream(target: { userId?: string; groupId?: string }, stream: AsyncIterable<string>): Promise<void> {
    // 流式推送：先发送"正在输入"，然后逐段更新消息
    let lastMessageId: string | null = null;
    let buffer = "";

    for await (const chunk of stream) {
      buffer += chunk;
      if (buffer.length >= 50) { // 每 50 字符更新一次
        lastMessageId = await this.updateOrSend(target, buffer, lastMessageId);
        buffer = "";
      }
    }
    if (buffer) {
      await this.updateOrSend(target, buffer, lastMessageId);
    }
  }

  disconnect(): Promise<void> {
    return this.client?.close();
  }
}
```
