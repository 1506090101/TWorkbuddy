# F2.5 — 代码知识图谱

> **Phase**: Phase 2 — Agent 编排 + 代码理解
> **Priority**: P0
> **Status**: Not Started
> **Dependencies**: F1.9 (项目自动感知)

## 概述

代码知识图谱是 Phase 2 中最复杂也最有价值的功能。其核心思想是：预先将项目代码通过 tree-sitter 解析为 AST，提取符号定义（函数/类/方法/接口等）和符号关系（调用/导入/继承/实现等），以图谱（节点+边+摘要）形式存储在 SQLite 中。Agent 查询代码时不再需要读取整个文件，而是先查图谱获取结构和关系摘要，只在需要具体实现时才按行范围读取代码段。

这种"图谱优先"的方式相比传统"Agent 直接读文件"可节省 90%+ token。例如理解一个函数实现，传统方式读取整个文件约 3000 tokens，知识图谱方式只需查摘要+按需读段约 300 tokens。在大型项目代码审查场景中，传统方式可能超出上下文窗口无法完成，而知识图谱方式仅需约 2000 tokens 即可完成。

图谱支持增量更新——文件保存时自动重新解析变化文件；支持渐进式构建——打开项目 1 秒内可用，后台逐步完成全量索引；AST 解析在 Worker 线程运行不阻塞 UI；摘要生成使用可配置的 LLM 模型批量异步处理。

```typescript
/** 文件节点 */
interface FileNode {
  id: string;                      // 文件路径 hash
  path: string;                    // 相对项目根路径
  language: string;                // typescript, python, rust...
  lines: number;                   // 总行数

  summary: string;                 // AI 生成的文件摘要（1-3 句话）
  exports: string[];               // 导出的符号 ID 列表
  imports: string[];               // 依赖的文件 ID 列表
  hash: string;                    // 文件内容 hash（增量更新判断）
}

/** 符号类型 */
type SymbolType = "function" | "class" | "method" | "variable" | "interface" | "type" | "enum" | "constant";

/** 符号节点 */
interface SymbolNode {
  id: string;                      // `${filePath}:${name}:${type}`
  name: string;                   // 符号名
  type: SymbolType;
  filePath: string;               // 所属文件
  location: {
    startLine: number;
    endLine: number;
    startColumn: number;
    endColumn: number;
  };
  signature: string;               // 函数签名 / 类定义头
  modifiers: string[];             // public, private, static, async, export...
  docstring?: string;              // JSDoc / docstring / 注释
  summary: string;                 // AI 生成的符号摘要
  complexity?: number;             // 圈复杂度（函数）
  parameters?: ParamInfo[];        // 函数参数
  returnType?: string;             // 返回类型
}

interface ParamInfo {
  name: string;
  type: string;
  defaultValue?: string;
  optional: boolean;
}

/** 关系边类型 */
type EdgeType = "CALLS" | "IMPORTS" | "EXPORTS" | "EXTENDS" | "IMPLEMENTS" | "REFERENCES" | "DEPENDS_ON" | "OVERRIDES";

interface RelationshipEdge {
  id: string;
  type: EdgeType;
  source: string;                  // 源节点 ID（符号或文件）
  target: string;                  // 目标节点 ID
  sourceLocation?: {
    line: number;
    column: number;

  metadata?: Record<string, any>;   // 额外信息
}
```

### 关键实现逻辑

**索引构建流程**:

```typescript
// main/services/knowledge-graph/index-builder.ts

import { Worker } from "worker_threads";
import * as path from "path";
import { FileNode, SymbolNode, RelationshipEdge, IndexProgress } from "../../../shared/types/knowledge-graph";

export class IndexBuilder {
  private worker: Worker;
  private db: Database;
  private progress: IndexProgress = {
    phase: 0, totalFiles: 0, indexedFiles: 0,
    totalSymbols: 0, totalEdges: 0,
    status: "scanning", message: "Scanning files...",
  };

  /**
   * 渐进式索引构建
   * Phase 0: 目录扫描 (<1秒) → 立即可用
   * Phase 1: 入口文件解析 (~5秒) → 基本可用
   * Phase 2: 全量解析 (后台) → 完整图谱
   * Phase 3: 摘要生成 (最低优先级) → 完全理解
   */
  async buildIndex(projectPath: string): Promise<void> {
    // Phase 0: 扫描目录结构
    this.progress.phase = 0;
    this.progress.status = "scanning";
    const files = await this.scanFiles(projectPath);
    this.progress.totalFiles = files.length;
    this.notifyProgress();

    // 按优先级排序：入口 > 核心模块 > 业务文件 > 测试 > 配置
    const sortedFiles = this.sortByPriority(files);

    // Phase 1: 解析入口文件 + 直接依赖
    this.progress.phase = 1;
    this.progress.status = "indexing";
    const entryFiles = sortedFiles.filter(f => this.isEntryPoint(f));
    await this.indexFiles(entryFiles, projectPath);
    this.notifyProgress();

    // Phase 2: 后台渐进式解析所有文件
    this.progress.phase = 2;
    const remainingFiles = sortedFiles.filter(f => !entryFiles.includes(f));
    await this.indexFiles(remainingFiles, projectPath, { background: true });

    // Phase 3: 摘要生成（批量异步，最低优先级）
    this.progress.phase = 3;
    this.progress.status = "summarizing";
    await this.generateSummaries(projectPath);

    this.progress.status = "complete";
    this.progress.message = "Indexing complete";
    this.notifyProgress();
  }

  /** 扫描文件（递归遍历 + .gitignore 过滤） */
  private async scanFiles(projectPath: string): Promise<string[]> {
    const gitignore = await this.parseGitignore(projectPath);
    const ignorePatterns = [
      ...gitignore,
      "node_modules/", "dist/", ".git/", "build/",
      "*.min.js", "*.map",
    ];
    
    const files: string[] = [];
    await this.walkDir(projectPath, "", ignorePatterns, files);
    return files;
  }

  /** 索引文件（通过 Worker 线程解析 AST） */
  private async indexFiles(
    files: string[],
    projectPath: string,
    opts?: { background?: boolean }
  ): Promise<void> {
    for (const file of files) {
      if (opts?.background) {
        // 后台索引，允许被查询插队
        await this.indexFileWithInterruption(file, projectPath);
      } else {
        await this.indexFile(file, projectPath);
      }
      this.progress.indexedFiles++;
      this.notifyProgress();
    }
  }

  /** 索引单个文件 */
  private async indexFile(relativePath: string, projectPath: string): Promise<void> {
    const fullPath = path.join(projectPath, relativePath);
    const content = await fs.readFile(fullPath, "utf-8");
    const hash = crypto.createHash("sha256").update(content).digest("hex");
    
    // 检查是否需要更新
    const existing = await this.db.getFileNode(relativePath);
    if (existing && existing.hash === hash) return;

    // 发送到 Worker 线程解析 AST
    const result = await this.sendToWorker({
      type: "parse-file",
      filePath: relativePath,
      content,
      language: this.detectLanguage(relativePath),
    });

    // 删除旧节点和关联边
    await this.db.deleteSymbolsByFile(relativePath);
    await this.db.deleteEdgesByFile(relativePath);

    // 更新文件节点
    await this.db.upsertFileNode({
      id: hashFile(relativePath),
      path: relativePath,
      language: result.language,
      lines: content.split("\n").length,
      lastModified: (await fs.stat(fullPath)).mtimeMs,
      lastIndexed: Date.now(),
      hash,
      summary: existing?.summary || "", // 保留旧摘要
    });

    // 写入新符号和边
    await this.db.insertSymbols(result.symbols);
    await this.db.insertEdges(result.edges);
    
    // 更新 FTS5 索引
    await this.db.updateFTSIndex(result.symbols);

    this.progress.totalSymbols += result.symbols.length;
    this.progress.totalEdges += result.edges.length;
  }
}
```

**Worker 线程（AST 解析）**:

```typescript
// main/services/knowledge-graph/code-graph-worker.ts

import { parentPort, Worker } from "worker_threads";
import Parser from "tree-sitter";
import TypeScript from "tree-sitter-typescript";
import Python from "tree-sitter-python";
// import Rust from "tree-sitter-rust";
// import Go from "tree-sitter-go";
// import Java from "tree-sitter-java";

const parsers: Record<string, any> = {
  typescript: new Parser().setLanguage(TypeScript.typescript),
  tsx: new Parser().setLanguage(TypeScript.tsx),
  javascript: new Parser().setLanguage(TypeScript.typescript),
  python: new Parser().setLanguage(Python),
};

parentPort?.on("message", async (msg: { type: string; filePath: string; content: string; language: string }) => {
  if (msg.type === "parse-file") {
    const result = await parseFile(msg.filePath, msg.content, msg.language);
    parentPort?.postMessage({ type: "parsed", ...result });
  }
});

async function parseFile(filePath: string, content: string, language: string) {
  const parser = parsers[language];
  if (!parser) {
    return { language, symbols: [], edges: [] };
  }

  const tree = parser.parse(content);
  const symbols = extractSymbols(tree.rootNode, filePath);
  const edges = extractRelationships(tree.rootNode, filePath, symbols);

  // 计算圈复杂度
  for (const sym of symbols) {
    if (sym.type === "function" || sym.type === "method") {
      sym.complexity = calculateComplexity(tree.rootNode, sym);
    }
  }

  return { language, symbols, edges };
}

/** 提取符号定义 */
function extractSymbols(node: any, filePath: string): SymbolNode[] {
  const symbols: SymbolNode[] = [];
  
  function walk(n: any) {
    const symbolType = mapNodeType(n.type);
    if (symbolType) {
      symbols.push({
        id: `${filePath}:${n.childForFieldName("name")?.text}:${symbolType}`,
        name: n.childForFieldName("name")?.text || "anonymous",
        type: symbolType,
        filePath,
        location: {
          startLine: n.startPosition.row + 1,
          endLine: n.endPosition.row + 1,
          startColumn: n.startPosition.column,
          endColumn: n.endPosition.column,
        },
        signature: extractSignature(n),
        modifiers: extractModifiers(n),
        docstring: extractDocstring(n, content),
        summary: "", // 异步生成
        parameters: extractParameters(n),
        returnType: extractReturnType(n),
      });
    }
    for (let i = 0; i < n.childCount; i++) {
      walk(n.child(i));
    }
  }
  walk(node);
  return symbols;
}
```

**增量更新机制**:

```typescript
// main/services/knowledge-graph/incremental-updater.ts

export class IncrementalUpdater {
  /**
   * 文件变更时触发增量更新
   * 触发时机：Agent edit/write 工具执行后、git checkout、项目重新打开
   */
  async onFileChanged(filePath: string, projectPath: string): Promise<void> {
    const fullPath = path.join(projectPath, filePath);
    const newHash = hashFile(await fs.readFile(fullPath, "utf-8"));
    const existing = await this.db.getFileNode(filePath);
    
    if (existing && existing.hash === newHash) return; // 未变化

    // 1. 删除旧节点和关联边
    await this.db.deleteSymbolsByFile(filePath);
    await this.db.deleteEdgesByFile(filePath);

    // 2. 重新解析该文件
    const { symbols, edges } = await this.parseWithTreeSitter(filePath, projectPath);

    // 3. 更新文件节点
    await this.db.upsertFileNode({
      path: filePath,
      hash: newHash,
      lastIndexed: Date.now(),
    });

    // 4. 写入新符号和边
    await this.db.insertSymbols(symbols);
    await this.db.insertEdges(edges);

    // 5. 异步更新摘要（如果文件变化较大）
    if (shouldRegenerateSummary(existing, newHash)) {
      this.queueSummaryUpdate(filePath);
    }

    // 6. 更新跨文件引用（其他文件引用了此文件的符号）
    await this.updateCrossFileReferences(filePath);
  }
}
```

**摘要生成策略**:

```typescript
// main/services/knowledge-graph/summary-generator.ts

export class SummaryGenerator {
  /**
   * 摘要生成策略：
   * - 入口文件：详细摘要 + 导出 API 列表
   * - 核心业务文件：符号级摘要 + 文件级摘要
   * - 工具/辅助文件：仅文件级摘要
   * - 测试文件：仅文件级摘要
   * - 配置文件：跳过
   * - 大文件 (>500行)：分段摘要 + 合并
   * - 第三方依赖：跳过
   */
  async generateSummaries(projectPath: string): Promise<void> {
    const files = await this.db.getAllFiles();
    
    for (const file of files) {
      // 跳过配置文件
      if (this.isConfigFile(file.path)) continue;
      
      // 文件级摘要
      const fileSummary = await this.generateFileSummary(file);
      await this.db.updateFileSummary(file.path, fileSummary);
      
      // 符号级摘要（仅核心文件）
      if (this.isCoreFile(file)) {
        const symbols = await this.db.getSymbolsByFile(file.path);
        const exportedSymbols = symbols.filter(s => s.modifiers.includes("export"));
        
        for (const sym of exportedSymbols) {
          const symSummary = await this.generateSymbolSummary(sym, file);
          await this.db.updateSymbolSummary(sym.id, symSummary);
        }
      }
    }
  }

  /** 使用 LLM 生成文件摘要 */
  private async generateFileSummary(file: FileNode): Promise<string> {
    const prompt = `请用 1-3 句话描述以下文件的职责和功能。只返回摘要，不要多余内容。

文件: ${file.path}
语言: ${file.language}
行数: ${file.lines}

文件内容（前100行）:
${fileContent.slice(0, 100)}
`;
    const result = await this.llmClient.complete({
      model: this.config.summaryModel || "gpt-4o-mini",
      messages: [{ role: "user", content: prompt }],
      maxTokens: 200,
    });
    return result.content;
  }
}
```
