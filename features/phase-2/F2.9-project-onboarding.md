# F2.9 — 项目首次接入引导

> **Phase**: Phase 2 — Agent 编排 + 代码理解
> **Priority**: P1
> **Status**: Not Started
> **Dependencies**: F1.9 (项目自动感知), F2.5 (代码知识图谱)

## 概述

项目首次接入引导功能解决了"用户打开新项目时不知道 Agent 对项目了解到什么程度"的体验问题。当用户首次打开一个项目时，系统在 1 秒内完成项目检测（技术栈/目录结构/入口文件/构建命令/测试命令），然后后台渐进式构建知识图谱，同时生成项目卡片展示架构概览图、编码约定、关键文件清单和依赖分析，最后给出首轮对话建议。

项目卡片在侧边栏展示，包含技术栈标签、架构概览图（可交互的模块关系图）、文件统计、索引进度和编码约定摘要。编码约定从 eslint/prettier/tsconfig 配置中自动提取，Agent 在生成代码时自动遵循这些约定。这个引导流程让用户打开项目后立即知道"Agent 已经理解了什么"和"可以开始做什么"，消除冷启动的困惑。

- [ ] 项目卡片包含：技术栈标签

- [ ] 编码约定提取：从 .prettierrc 提取格式化配置
- [ ] 编码约定提取：从 tsconfig.json 提取 typeStrictness

- [ ] 编码约定提取：检测 commit 规范（conventional commits 等）

```typescript
/** 项目卡片数据 */
interface ProjectCard {
  projectId: string;
  projectPath: string;
  projectName: string;
  
  // 基础信息（来自 F1.9 项目感知）
  techStack: string[];               // ["TypeScript", "React", "Express", "PostgreSQL"]
  language: string;
  framework?: string;
  packageManager?: string;
  buildSystem?: string;
  testCommand?: string;
  lintCommand?: string;
  
  // 统计信息
  fileStats: {
    totalFiles: number;
    totalLines: number;
    languageDistribution: { language: string; percentage: number }[];
    indexedFiles: number;            // 知识图谱已索引文件数
  };
  
  // 关键文件
  keyFiles: {
    entryPoints: string[];          // ["src/index.ts", "src/app.ts"]
    configFiles: string[];          // ["package.json", "tsconfig.json", ".eslintrc"]
    coreModules: string[];         // 从知识图谱 topFiles 生成
  };
  
  // 架构概览
  architecture: {
    modules: ModuleNode[];          // 模块节点列表
    dependencies: { source: string; target: string }[];  // 模块依赖关系
    entryPoints: string[];          // 从知识图谱获取
  };
  
  // 编码约定
  conventions: CodingConventions;
  
  // 依赖分析
  dependencies: {
    direct: DependencyInfo[];
    devDependencies: DependencyInfo[];
    riskyDeps?: string[];           // 潜在风险依赖（过时/有安全漏洞）
  };
  
  // 首轮对话建议
  suggestedQuestions: string[];
  
  createdAt: number;
  updatedAt: number;
}

/** 模块节点 */
interface ModuleNode {
  id: string;                       // 目录路径
  name: string;
  path: string;
  summary: string;                  // 模块摘要（从知识图谱）
  fileCount: number;
  symbolCount: number;
}

/** 编码约定 */
interface CodingConventions {
  language: string;                // "typescript"
  styleGuide: {
    indent: "2spaces" | "4spaces" | "tab";
    quotes: "single" | "double";
    semicolons: boolean;
    trailingComma: "none" | "es5" | "all";
    namingConvention: {
      variables: string;            // "camelCase"
      classes: string;             // "PascalCase"
      constants: string;           // "UPPER_SNAKE"
      files: string;               // "kebab-case" | "camelCase" | "PascalCase"
    };
  };
  typeStrictness: "strict" | "moderate" | "loose";
  testFramework?: string;           // "jest" | "vitest" | "pytest"
  commitConvention?: string;        // "conventional" | "custom"
}

/** 依赖信息 */
interface DependencyInfo {
  name: string;
  version: string;
  latestVersion?: string;
  isOutdated?: boolean;
  hasVulnerability?: boolean;
}

/** 引导状态 */
interface OnboardingState {
  projectId: string;
  status: "detecting" | "indexing" | "ready";
  progress?: {
    phase: 0 | 1 | 2 | 3;
    indexed: number;
    total: number;
    message: string;
  };
  isFirstTime: boolean;
}
```

### 关键实现逻辑

**接入引导流程**:

```typescript
// main/services/onboarding-manager.ts

import { ProjectContext, CodingConventions } from "../../shared/types/onboarding";

export class OnboardingManager {
  /**
   * 启动项目接入引导
   * 1. 项目检测 (<1秒)
   * 2. 知识图谱渐进构建 (后台)
   * 3. 生成项目卡片 (Phase 1 完成后)
   * 4. 建议首轮对话
   */
  async startOnboarding(projectPath: string): Promise<{ projectId: string }> {
    const projectId = hashPath(projectPath);
    
    // 1. 项目检测（复用 F1.9 项目感知）
    const context = await this.projectDetector.detect(projectPath);
    
    // 通知渲染进程：检测完成
    this.notifyProgress({
      projectId,
      status: "indexing",
      progress: { phase: 0, indexed: 0, total: 0, message: "Scanning files..." },
      isFirstTime: await this.isFirstTime(projectId),
    });

    // 2. 启动知识图谱索引（复用 F2.5）
    this.knowledgeGraph.buildIndex(projectPath).then(async () => {
      // 3. 索引完成后生成项目卡片
      const card = await this.generateProjectCard(projectId, projectPath, context);
      await this.projectCardStore.save(card);
      
      // 4. 生成首轮对话建议
      const suggestions = await this.generateSuggestions(context, card);
      
      this.notifyProgress({
        projectId,
        status: "ready",
        isFirstTime: await this.isFirstTime(projectId),
      });
    });

    // 知识图谱进度事件转发
    this.knowledgeGraph.onProgress((progress) => {
      this.notifyProgress({
        projectId,
        status: "indexing",
        progress,
        isFirstTime: false,
      });
    });

    return { projectId };
  }

  /** 生成项目卡片 */
  private async generateProjectCard(
    projectId: string,
    projectPath: string,
    context: ProjectContext
  ): Promise<ProjectCard> {
    // 从知识图谱获取项目概览
    const overview = await this.knowledgeGraph.getProjectOverview(projectId);
    
    // 从知识图谱获取 topFiles 作为核心模块
    const coreModules = overview.topFiles.map(f => ({
      path: f.path,
      summary: f.summary,
      incomingRefs: f.incomingRefs,
    }));

    // 提取编码约定
    const conventions = await this.extractConventions(projectPath);

    // 分析依赖
    const dependencies = await this.analyzeDependencies(context);

    // 生成架构概览
    const architecture = await this.generateArchitectureOverview(projectId);

    // 生成首轮对话建议
    const suggestedQuestions = this.generateSuggestedQuestions(context, overview);

    return {
      projectId,
      projectPath,
      projectName: path.basename(projectPath),
      techStack: this.buildTechStack(context),
      language: context.language,
      framework: context.framework,
      packageManager: context.packageManager,
      buildSystem: context.buildSystem,
      testCommand: context.testCommand,
      lintCommand: context.lintCommand,
      fileStats: {
        totalFiles: overview.totalFiles,
        totalLines: await this.countTotalLines(projectPath),
        languageDistribution: await this.getLanguageDistribution(projectPath),
        indexedFiles: overview.totalFiles,
      },
      keyFiles: {
        entryPoints: context.structure.entryPoints,
        configFiles: context.structure.configFiles,
        coreModules: coreModules.map(m => m.path),
      },
      architecture,
      conventions,
      dependencies,
      suggestedQuestions,
      createdAt: Date.now(),
      updatedAt: Date.now(),
    };
  }

  /** 提取编码约定 */
  private async extractConventions(projectPath: string): Promise<CodingConventions> {
    const conventions: CodingConventions = {
      language: "typescript",
      styleGuide: {
        indent: "2spaces",
        quotes: "single",
        semicolons: true,
        trailingComma: "all",
        namingConvention: {
          variables: "camelCase",
          classes: "PascalCase",
          constants: "UPPER_SNAKE",
          files: "kebab-case",
        },
      },
      typeStrictness: "moderate",
    };

    // 从 .prettierrc 提取
    const prettierConfig = await this.tryReadJSON(projectPath, ".prettierrc");
    if (prettierConfig) {
      if (prettierConfig.useTabs) conventions.styleGuide.indent = "tab";
      else if (prettierConfig.tabWidth === 4) conventions.styleGuide.indent = "4spaces";
      
      if (prettierConfig.singleQuote === false) conventions.styleGuide.quotes = "double";
      if (prettierConfig.semi === false) conventions.styleGuide.semicolons = false;
      if (prettierConfig.trailingComma) conventions.styleGuide.trailingComma = prettierConfig.trailingComma;
    }

    // 从 .eslintrc 提取
    const eslintConfig = await this.tryReadJSON(projectPath, ".eslintrc.json")
      ?? await this.tryReadJSON(projectPath, ".eslintrc");
    if (eslintConfig?.rules) {
      const rules = eslintConfig.rules;
      if (rules.indent) {
        const indentRule = Array.isArray(rules.indent) ? rules.indent[1] : rules.indent;
        if (indentRule === "tab") conventions.styleGuide.indent = "tab";
        else if (indentRule === 4) conventions.styleGuide.indent = "4spaces";
      }
      if (rules.quotes) {
        const quotesRule = Array.isArray(rules.quotes) ? rules.quotes[1] : rules.quotes;
        if (quotesRule === "double") conventions.styleGuide.quotes = "double";
      }
      if (rules.semi) {
        const semiRule = Array.isArray(rules.semi) ? rules.semi[1] : rules.semi;
        conventions.styleGuide.semicolons = semiRule !== "never";
      }
    }

    // 从 tsconfig.json 提取严格度
    const tsConfig = await this.tryReadJSON(projectPath, "tsconfig.json");
    if (tsConfig?.compilerOptions) {
      const opts = tsConfig.compilerOptions;
      if (opts.strict === true) {
        conventions.typeStrictness = "strict";
      } else if (opts.strictNullChecks || opts.strictFunctionTypes) {
        conventions.typeStrictness = "moderate";
      } else {
        conventions.typeStrictness = "loose";
      }
    }

    // 检测测试框架
    const packageJson = await this.tryReadJSON(projectPath, "package.json");
    if (packageJson) {
      const allDeps = { ...packageJson.dependencies, ...packageJson.devDependencies };
      if (allDeps["jest"]) conventions.testFramework = "jest";
      else if (allDeps["vitest"]) conventions.testFramework = "vitest";
      else if (allDeps["mocha"]) conventions.testFramework = "mocha";
      else if (allDeps["pytest"] || allDeps["pytest"]) conventions.testFramework = "pytest";
    }

    // 检测 commit 规范
    const commitlintConfig = await this.tryReadJSON(projectPath, ".commitlintrc.json");
    if (commitlintConfig) {
      conventions.commitConvention = "conventional";
    }

    return conventions;
  }

  /** 生成首轮对话建议 */
  private generateSuggestedQuestions(
    context: ProjectContext,
    overview: ProjectOverview
  ): string[] {
    const suggestions: string[] = [];
    
    // 基于核心模块
    if (overview.topFiles.length > 0) {
      const topFile = overview.topFiles[0];
      suggestions.push(`这个项目的 ${topFile.path} 是做什么的？`);
    }

    // 基于入口文件
    if (context.structure.entryPoints.length > 0) {
      suggestions.push(`帮我梳理 ${context.structure.entryPoints[0]} 的执行流程`);
    }

    // 基于技术栈
    if (context.framework) {
      suggestions.push(`这个项目的 ${context.framework} 架构是怎么组织的？`);
    }

    // 基于测试
    if (context.testCommand) {
      suggestions.push(`这个项目的测试覆盖率怎么样？`);
    } else {
      suggestions.push(`帮我为这个项目添加单元测试`);
    }

    // 通用建议
    suggestions.push(`帮我找到所有处理用户认证的代码`);
    suggestions.push(`这个项目有哪些潜在的技术债务？`);

    return suggestions.slice(0, 5);
  }
}
```

## UI/UX 规范

- **项目卡片位置**：侧边栏顶部，项目打开后自动展示
- **进度指示器**：卡片顶部显示接入阶段和进度条，"正在理解你的项目... 45/120 文件已索引"
- **技术栈标签**：圆角标签，每个技术一个标签，使用对应品牌色
- **架构概览图**：可交互的模块关系图，节点表示模块（大小=文件数），连线表示依赖，点击节点跳转
- **编码约定**：简洁的信息列表，图标+规则名+值
- **建议问题**：圆角卡片列表，每条建议可点击直接填入输入框发送
- **首次 vs 再次**：首次打开显示完整卡片+进度+建议；再次打开显示简略卡片（可展开详情）

## 测试要点

- **单元测试**: 编码约定提取（eslint/prettier/tsconfig 解析）；依赖分析；建议生成逻辑
- **集成测试**: 完整接入流程（打开项目→检测→索引→卡片生成→建议展示）；首次 vs 再次打开区分
- **E2E 测试**: 用户打开新项目→看到进度指示器→索引完成→查看项目卡片→点击建议问题→Agent 回答

## 备注

- 项目检测复用 F1.9 项目自动感知的 ProjectDetector
- 知识图谱渐进构建复用 F2.5 的 IndexBuilder
