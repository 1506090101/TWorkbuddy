# F7.6 — 插件生态与分享

> **Phase**: Phase 7 — 精细化 (Polish)
> **Priority**: P2
> **Status**: Not Started
> **Dependencies**: F7.1 (Skills 系统), F7.2 (自定义工具脚本), F2.1 (Agent 定义模型)

## 概述

F7.1 Skills 系统和 F7.2 自定义工具脚本提供了扩展能力，F2.1 Agent 定义支持导出/导入，但这些功能缺少发现机制——用户得手动分享文件。F7.6 插件生态与分享构建一个社区市场索引，支持一键安装、打包格式（`.agentbuddy-pack`）、兼容性检查和安装安全审计，让 Skills/Agents/Tools/Packs 的发现、安装和分享变得无缝。

社区索引采用轻量级方案（基于 GitHub 仓库或 JSON API），不需要自建中央服务器。打包格式 `.agentbuddy-pack`（实际为 zip）包含 manifest.json 元数据 + agents/ + skills/ + tools/ + README.md。安装前显示权限请求，自定义工具脚本沙箱执行，社区评分和评论机制，官方认证标记（verified）。

## PRD 对应

- **3.16.8 插件生态与分享 (P2)**: 社区索引（listPlugins/install/update/publish）、打包格式（.agentbuddy-pack）、兼容性检查、安装安全审计

## 验收标准

### 功能验收
- [ ] 社区索引：`listPlugins` 浏览插件（按 type/category/sort/search 过滤）
- [ ] 一键安装：`install(pluginId)` 下载到 `~/.agentbuddy/{type}s/` → 验证签名 → 注册
- [ ] 更新检查：`checkUpdates` 批量检查已安装插件的更新
- [ ] 单个更新：`update(pluginId)` 更新指定插件
- [ ] 发布插件：`publish(pluginPath)` 打包并上传到社区仓库
- [ ] 兼容性检查：`checkCompatibility(pluginId)` 检查模型要求、工具依赖、版本兼容
- [ ] 打包格式 `.agentbuddy-pack`：zip 包含 manifest.json + agents/ + skills/ + tools/ + README.md
- [ ] 安装前权限请求：显示插件会使用哪些工具、访问哪些目录
- [ ] 自定义工具脚本沙箱执行（受限权限）
- [ ] 社区评分和评论机制
- [ ] 官方认证标记（verified）
- [ ] 插件管理界面（浏览/安装/卸载/更新）

### 非功能验收
- [ ] 插件安装后不影响已有功能（隔离性）
- [ ] 恶意插件检测（扫描危险 API 调用）

interface PluginEntry {
  id: string;
  type: "skill" | "agent" | "tool" | "pack";
  name: string;
  description: string;
  author: string;
  version: string;
  category: string;
  tags: string[];
  rating: number;                  // 平均评分
  ratingCount: number;
  downloadCount: number;
  verified: boolean;              // 官方认证
  publishedAt: number;
  updatedAt: number;
  downloadUrl: string;
  manifestUrl: string;
  screenshots?: string[];
  readmeUrl: string;
}

// 打包格式 manifest.json
interface PluginManifest {
  name: string;
  version: string;
  author: string;
  description: string;
  type: "skill" | "agent" | "tool" | "pack";
  category: string;
  tags: string[];
  
  // 兼容性要求
  requirements: {
    minAppVersion?: string;        // 最低 Agent Buddy 版本
    requiredTools?: string[];     // 依赖的工具
    requiredModels?: string[];    // 建议的模型
    requiredProviders?: string[]; // 建议的 Provider 类型
  };
  
  // 权限声明
  permissions: {
    fileAccess: string[];          // 需要访问的目录
    tools: string[];               // 会使用的工具
    network: boolean;              // 是否需要网络访问
  };
  
  // 包含的资源
  contents: {
    agents?: string[];
    skills?: string[];
    tools?: string[];
  };
}

// 安装结果
interface InstallResult {
  success: boolean;
  installedPath: string;
  errors?: string[];
  warnings?: string[];
}

// 兼容性报告
interface CompatibilityReport {
  compatible: boolean;
  issues: {
    type: "version" | "tool" | "model" | "provider";
    message: string;
    severity: "error" | "warning";
  }[];
}

// .agentbuddy-pack 文件结构（zip）：

### 关键实现逻辑

**插件安装流程**：

```typescript
// main/services/marketplace/PluginInstaller.ts

class PluginInstaller {
  // 安装插件
  async install(pluginId: string): Promise<InstallResult> {
    // 1. 获取插件信息
    const entry = await this.fetchPluginEntry(pluginId);

    // 2. 兼容性检查
    const compat = await this.checkCompatibility(pluginId);
    if (!compat.compatible) {
      return {
        success: false,
        installedPath: "",
        errors: compat.issues.filter(i => i.severity === "error").map(i => i.message),
      };
    }

    // 3. 下载 .agentbuddy-pack
    const packPath = await this.downloadPack(entry.downloadUrl);

    // 4. 验证签名
    const verified = await this.verifySignature(packPath);
    if (!verified) {
      return { success: false, installedPath: "", errors: ["Signature verification failed"] };
    }

    // 5. 解压并读取 manifest
    const manifest = await this.readManifest(packPath);

    // 6. 安全审计（扫描危险 API）
    const audit = await this.securityAudit(packPath, manifest);
    if (audit.blocked) {
      return { success: false, installedPath: "", errors: audit.issues };
    }

    // 7. 安装到对应目录
    const installPath = await this.installFiles(packPath, manifest);

    // 8. 注册插件
    await this.registerPlugin(manifest, installPath);

    return {
      success: true,
      installedPath: installPath,
      warnings: audit.warnings,
    };
  }

  // 从 .agentbuddy-pack 安装文件
  async installFromFile(packPath: string): Promise<InstallResult> {
    // 1. 解压
    const tempDir = await this.extractPack(packPath);

    // 2. 读取 manifest
    const manifest: PluginManifest = JSON.parse(
      await fs.readFile(path.join(tempDir, "manifest.json"), "utf-8")
    );

    // 3. 安全审计
    const audit = await this.securityAudit(tempDir, manifest);

    // 4. 安装各类型文件
    if (manifest.contents.agents) {
      for (const agent of manifest.contents.agents) {
        await fs.copyFile(
          path.join(tempDir, "agents", agent),
          path.join(os.homedir(), ".agentbuddy/agents", agent)
        );
      }
    }

    if (manifest.contents.skills) {
      for (const skill of manifest.contents.skills) {
        await fs.copyFile(
          path.join(tempDir, "skills", skill),
          path.join(os.homedir(), ".agentbuddy/skills", skill)
        );
      }
    }

    if (manifest.contents.tools) {
      for (const tool of manifest.contents.tools) {
        await fs.copyFile(
          path.join(tempDir, "tools", tool),
          path.join(os.homedir(), ".agentbuddy/tools", tool)
        );
      }
    }

    return { success: true, installedPath: path.join(os.homedir(), ".agentbuddy") };
  }

  // 创建 .agentbuddy-pack
  async createPack(opts: {
    name: string;
    agents?: string[];
    skills?: string[];
    tools?: string[];
  }): Promise<string> {
    const tempDir = path.join(os.tmpdir(), `agentbuddy-pack-${Date.now()}`);
    await fs.mkdir(tempDir, { recursive: true });

    // 1. 生成 manifest.json
    const manifest: PluginManifest = {
      name: opts.name,
      version: "1.0.0",
      author: "user",
      description: "",
      type: "pack",
      category: "general",
      tags: [],
      requirements: {},
      permissions: { fileAccess: [], tools: [], network: false },
      contents: {
        agents: opts.agents,
        skills: opts.skills,
        tools: opts.tools,
      },
    };
    await fs.writeFile(path.join(tempDir, "manifest.json"), JSON.stringify(manifest, null, 2));

    // 2. 复制 agents/skills/tools
    if (opts.agents) {
      await fs.mkdir(path.join(tempDir, "agents"), { recursive: true });
      for (const agent of opts.agents) {
        await fs.copyFile(
          path.join(os.homedir(), ".agentbuddy/agents", agent),
          path.join(tempDir, "agents", agent)
        );
      }
    }
    // ... skills, tools 类似

    // 3. 打包为 zip
    const packPath = path.join(os.homedir(), "Downloads", `${opts.name}.agentbuddy-pack`);
    await zipDirectory(tempDir, packPath);

    return packPath;
  }

  // 安全审计
  private async securityAudit(
    packPath: string,
    manifest: PluginManifest
  ): Promise<{ blocked: boolean; issues: string[]; warnings: string[] }> {
    const issues: string[] = [];
    const warnings: string[] = [];

    // 扫描工具脚本中的危险 API
    if (manifest.contents.tools) {
      for (const tool of manifest.contents.tools) {
        const toolPath = path.join(packPath, "tools", tool);
        const content = await fs.readFile(toolPath, "utf-8");

        // 检查危险模式
        const dangerousPatterns = [
          /require\(['"]child_process['"]\)/,  // 执行系统命令
          /require\(['"]fs['"]\)\.unlink/,     // 删除文件
          /require\(['"]net['"]\)/,             // 网络访问（未声明）
          /eval\(/,                               // 动态执行
          /Function\(/,                          // 动态函数
        ];

        for (const pattern of dangerousPatterns) {
          if (pattern.test(content)) {
            issues.push(`Tool ${tool} contains potentially dangerous code: ${pattern}`);
          }
        }
      }
    }

    // 检查权限声明与实际使用是否一致
    if (!manifest.permissions.network) {
      // 检查是否有网络访问代码
    }

    return {
      blocked: issues.length > 0,
      issues,
      warnings,
    };
  }
}
```
