# F1.10 — Git 基础集成

> **Phase**: Phase 1 — 基础底座 + 代码安全
> **Priority**: P1
> **Status**: In Progress
> **Dependencies**: F1.3 (Pi SDK 集成)

## 概述

Agent 在辅助开发时，需要语义化地理解版本控制状态——不是通过 `bash` 运行 `git status` 然后解析文本输出，而是通过结构化的 API 直接获取分支信息、变更状态、提交历史等。本功能将 Git 操作封装为语义化的工具集，供 Agent 在工具循环中调用。

工具集涵盖三个层次：状态感知（gitStatus / gitBranch / gitLog）、差异理解（gitDiff / gitShow）、和基本操作（gitCommit / gitCreateBranch / gitCheckout / gitStash / gitStashPop）。每个工具返回结构化的 TypeScript 对象，而非原始 Git 命令输出，让 Agent 可以精确理解每一个字段含义。

关键设计包括：会话启动时自动执行 `gitStatus` 并将结果注入 Agent 上下文，Agent 立即知道当前在哪个分支、有哪些未提交的改动；提交信息可自动生成——Agent 根据 diff 内容调用 LLM 生成规范的 commit message；工具集成 F1.6 的权限确认机制（gitCommit 需要确认，只读操作自动放行）。

## V2 需求调整：先提供只读 Git 上下文

V2 首轮只将 Git 状态作为项目上下文：主进程以无 shell 子进程调用 Git，右侧项目面板展示分支和变更数，Agent 在每次工作开始时获得简洁状态摘要。提交、分支切换、stash、diff 与日志必须在变更审查能力成熟后逐项开放，不能借由通用 `bash` 绕过结构化 Git 边界。

### 关键实现逻辑

#### GitManager 核心

```typescript
// main/git/git-manager.ts

import { execSync } from "child_process";
import * as path from "path";

class GitManager {
  /**
   * 执行 Git 命令
   */
  private exec(args: string[], rootPath: string): string {
    try {
      return execSync(`git ${args.join(" ")}`, {
        cwd: rootPath,
        encoding: "utf-8",
        maxBuffer: 10 * 1024 * 1024, // 10MB
        timeout: 10000,
      }).trim();
    } catch (error: any) {
      // 未初始化仓库
      if (error.stderr?.includes("not a git repository")) {
        throw new GitError("NOT_A_REPO", "当前目录不是 Git 仓库");
      }
      throw error;
    }
  }

  /**
   * 获取 Git 状态
   */
  getStatus(rootPath: string): GitStatus {
    // 获取分支名
    const branch = this.exec(["rev-parse", "--abbrev-ref", "HEAD"], rootPath);

    // 获取 ahead/behind 信息
    let ahead = 0;
    let behind = 0;
    try {
      const tracking = this.exec(
        ["rev-list", "--left-right", "--count", `${branch}...@{upstream}`],
        rootPath,
      );
      [ahead, behind] = tracking.split("\t").map(Number);
    } catch {
      // 没有 upstream
    }

    // 解析 staged 变更
    const staged: FileChange[] = this.parseStatus(
      this.exec(["diff", "--cached", "--name-status"], rootPath),
      true,
    );

    // 解析 unstaged 变更
    const unstaged: FileChange[] = this.parseStatus(
      this.exec(["diff", "--name-status"], rootPath),
      false,
    );

    // 获取 untracked 文件
    const untracked = this.exec(
      ["ls-files", "--others", "--exclude-standard"],
      rootPath,
    )
      .split("\n")
      .filter(Boolean);

    return {
      branch,
      ahead,
      behind,
      staged,
      unstaged,
      untracked,
      totalChanges: staged.length + unstaged.length + untracked.length,
    };
  }

  /**
   * 解析 git status 输出
   */
  private parseStatus(output: string, staged: boolean): FileChange[] {
    if (!output) return [];

    return output.split("\n").map((line) => {
      const parts = line.split("\t");
      const statusCode = parts[0];

      let status: FileChange["status"];
      if (statusCode.startsWith("A")) status = "added";
      else if (statusCode.startsWith("D")) status = "deleted";
      else if (statusCode.startsWith("R")) status = "renamed";
      else status = "modified";

      return {
        path: parts[parts.length - 1],
        status,
      };
    });
  }

  /**
   * 获取 diff
   */
  getDiff(
    rootPath: string,
    opts?: {
      staged?: boolean;
      file?: string;
      commit1?: string;
      commit2?: string;
    },
  ): DiffResult {
    const args = ["diff", "--unified=3"];

    if (opts?.staged) args.push("--cached");
    if (opts?.commit1 && opts?.commit2) {
      args.push(opts.commit1, opts.commit2);
    }
    if (opts?.file) args.push("--", opts.file);

    const rawDiff = this.exec(args, rootPath);
    const files = this.parseDiff(rawDiff);

    return {
      staged: opts?.staged ?? false,
      files,
      rawDiff,
    };
  }

  /**
   * 解析 diff 输出为结构化数据
   */
  private parseDiff(rawDiff: string): FileDiff[] {
    if (!rawDiff) return [];

    const files: FileDiff[] = [];
    let currentFile: FileDiff | null = null;

    const lines = rawDiff.split("\n");

    for (const line of lines) {
      // 文件头: diff --git a/file b/file
      if (line.startsWith("diff --git ")) {
        if (currentFile) files.push(currentFile);
        currentFile = {
          path: "",
          status: "modified",
          additions: 0,
          deletions: 0,
          hunks: [],
          rawDiff: line + "\n",
        };
        continue;
      }

      if (!currentFile) continue;
      currentFile.rawDiff += line + "\n";

      // 文件路径提取
      if (line.startsWith("--- a/")) {
        currentFile.path = line.slice(6);
      }
      if (line.startsWith("+++ b/")) {
        if (!currentFile.path) currentFile.path = line.slice(6);
      }
      if (line.startsWith("new file")) currentFile.status = "added";
      if (line.startsWith("deleted file")) currentFile.status = "deleted";
      if (line.startsWith("rename from")) currentFile.status = "renamed";

      // 行计数
      if (line.startsWith("+") && !line.startsWith("+++")) {
        currentFile.additions++;
      }
      if (line.startsWith("-") && !line.startsWith("---")) {
        currentFile.deletions++;
      }
    }

    if (currentFile) files.push(currentFile);
    return files;
  }

  /**
   * 获取 commit 日志
   */
  getLog(
    rootPath: string,
    opts?: {
      limit?: number;
      file?: string;
      author?: string;
      since?: string;
    },
  ): CommitInfo[] {
    const limit = opts?.limit || 20;
    const format = "--format=%H|%h|%an|%at|%s";
    const args = ["log", format, `-${limit}`];

    if (opts?.file) args.push("--", opts.file);
    if (opts?.author) args.push("--author", opts.author);
    if (opts?.since) args.push("--since", opts.since);

    const output = this.exec(args, rootPath);
    if (!output) return [];

    return output.split("\n").map((line) => {
      const [hash, shortHash, author, date, ...messageParts] = line.split("|");
      return {
        hash,
        shortHash,
        author,
        date: parseInt(date) * 1000,
        message: messageParts.join("|"),
        files: [], // 后续可通过 git show --name-only 获取
      };
    });
  }

  /**
   * 提交
   */
  commit(
    rootPath: string,
    message: string,
    files?: string[],
  ): { hash: string } {
    const args = files
      ? ["commit", "-m", message, "--", ...files]
      : ["commit", "-m", message];

    const output = this.exec(args, rootPath);
    const match = output.match(/\[[\w/.-]+\s+([a-f0-9]+)\]/);
    const hash = match ? match[1] : "";

    return { hash };
  }

  /**
   * 创建分支
   */
  createBranch(rootPath: string, name: string, from?: string): void {
    const args = from
      ? ["checkout", "-b", name, from]
      : ["checkout", "-b", name];
    this.exec(args, rootPath);
  }

  /**
   * 切换分支
   */
  checkout(rootPath: string, branch: string): void {
    this.exec(["checkout", branch], rootPath);
  }

  /**
   * 暂存
   */
  stash(rootPath: string, message?: string): void {
    const args = message ? ["stash", "push", "-m", message] : ["stash"];
    this.exec(args, rootPath);
  }

  /**
   * 恢复暂存
   */
  stashPop(rootPath: string): void {
    this.exec(["stash", "pop"], rootPath);
  }

  /**
   * 构建 Agent 上下文注入字符串
   */
  buildContextString(status: GitStatus): string {
    const lines: string[] = [
      "## Git 状态",
      `- **当前分支**: \`${status.branch}\``,
    ];

    if (status.ahead > 0) lines.push(`- **领先远程**: ${status.ahead} commits`);
    if (status.behind > 0)
      lines.push(`- **落后远程**: ${status.behind} commits`);

    if (status.staged.length > 0) {
      lines.push(`- **已暂存**: ${status.staged.length} 个文件`);
      for (const f of status.staged) {
        lines.push(`  - \`${f.path}\` (${f.status})`);
      }
    }

    if (status.unstaged.length > 0) {
      lines.push(`- **未暂存**: ${status.unstaged.length} 个文件`);
      for (const f of status.unstaged) {
        lines.push(`  - \`${f.path}\` (${f.status})`);
      }
    }

    if (status.untracked.length > 0) {
      lines.push(`- **未跟踪**: ${status.untracked.length} 个文件`);
    }

    if (status.totalChanges === 0) {
      lines.push("- **工作区干净**，没有未提交的改动");
    }

    return lines.join("\n");
  }

  /**
   * 将 Git 操作注册为 Pi 工具
   */
  registerAsTools(rootPath: string): ToolDefinition[] {
    return [
      {
        name: "gitStatus",
        description: "获取 Git 仓库状态（分支、暂存/未暂存变更、未跟踪文件）",
        parameters: {},
        permission: "auto",
      },
      {
        name: "gitDiff",
        description: "获取 Git diff（工作区或暂存区）",
        parameters: {
          staged: { type: "boolean", description: "查看暂存区 diff" },
          file: { type: "string", description: "限定文件路径" },
        },
        permission: "auto",
      },
      {
        name: "gitLog",
        description: "获取 Git commit 历史",
        parameters: {
          limit: { type: "number", description: "返回数量限制，默认 20" },
          file: { type: "string", description: "限定文件" },
        },
        permission: "auto",
      },
      {
        name: "gitCommit",
        description: "提交暂存的改动",
        parameters: {
          message: {
            type: "string",
            description: "commit message",
            required: true,
          },
          files: { type: "array", description: "限定提交的文件" },
        },
        permission: "confirm", // 需要确认
      },
      {
        name: "gitCreateBranch",
        description: "创建并切换到新分支",
        parameters: {
          name: { type: "string", description: "新分支名", required: true },
          from: { type: "string", description: "基于哪个分支创建" },
        },
        permission: "confirm",
      },
      {
        name: "gitCheckout",
        description: "切换到指定分支",
        parameters: {
          branch: { type: "string", description: "目标分支名", required: true },
        },
        permission: "confirm",
      },
    ];
  }
}

class GitError extends Error {
  constructor(
    public code: string,
    message: string,
  ) {
    super(message);
    this.name = "GitError";
  }
}

export const gitManager = new GitManager();
```

// main/agent/context-builder.ts (修改)

async function buildInitialContext(rootPath: string): Promise<string> {
  const contexts: string[] = [];

  // Git 状态（F1.10）
  try {
    const gitStatus = gitManager.getStatus(rootPath);
    const gitContext = gitManager.buildContextString(gitStatus);
    contexts.push(gitContext);
  } catch (error: any) {
    if (error.code === "NOT_A_REPO") {
      contexts.push("## Git 状态\n- 当前目录未初始化 Git 仓库");
    }
  }

  // 项目上下文（F1.9）
  try {
    const projectContext = await projectDetector.detect(rootPath);
    const projectString = contextBuilder.buildSystemContext(projectContext);
    contexts.push(projectString);
  } catch {
    // 项目感知失败不影响 Agent 启动
  }

  return contexts.join("\n\n");
}

## 备注

- Git 工具使用 `child_process.execSync`，而非 `simple-git` 等第三方库（减少依赖）
- 所有 Git 操作限定在 `rootPath` 工作区内，不跨目录操作
- F3.2（Git 深度集成）将在此基础上添加 gitBlame / gitCreatePR / 合并冲突辅助
- Git 命令参数使用数组拼接，不使用模板字符串（防止路径注入）
