# F3.1 — MCP 客户端集成

> **Phase**: Phase 3 — MCP 工具生态 + 多项目 (MCP & Multi-Project)
> **Priority**: P0
> **Status**: Not Started
> **Dependencies**: F1.6 (内置工具集成与权限确认)

## 概述

MCP（Model Context Protocol）是一个开放协议，允许外部工具服务器以标准化方式向 LLM 暴露工具、资源和提示模板。Agent Buddy 作为一个面向开发者的多 Agent AI 平台，需要原生支持 MCP 工具生态——用户可以配置任意数量的 MCP Server，Agent Buddy 自动发现并注册其工具，让 Agent 能像使用内置工具一样调用 MCP 工具。

Pi SDK 核心不内置 MCP 支持（认为 MCP Server 会膨胀上下文），但通过扩展系统可以添加。Agent Buddy 的方案是构建一个 MCP 客户端扩展，将外部 MCP Server 的工具自动注册为 Pi 工具。关键设计是**按需加载**：不一次性注册所有 MCP Server 的全部工具定义，而是分两层——先注册轻量级工具目录（仅名称+描述，~10 tokens/tool），Agent 决定使用时才加载完整定义（含 schema，~200 tokens/tool）。这避免了 MCP 工具定义导致上下文膨胀的问题。

本功能还包含 MCP Resources 浏览 UI 和 MCP Prompts `/` 命令调用，以及在设置中心的 MCP Server 管理面板（添加/编辑/删除/测试连接）。

## 技术设计

### 数据模型

```typescript
// MCP Server 配置（存储在 ~/.agentbuddy/config/mcp.json）
interface McpServerConfig {
  name: string;                    // 唯一标识，如 "filesystem"
  command: string;                 // 启动命令，如 "npx"
  args: string[];                  // 命令参数，如 ["-y", "@modelcontextprotocol/server-filesystem"]
  env?: Record<string, string>;    // 环境变量，如 { "GITHUB_TOKEN": "ghp_xxx" }
  permission: "auto" | "confirm" | "disabled";  // 权限级别
  enabled: boolean;                // 是否启用
}

// mcp.json 文件格式
interface McpConfigFile {
  mcpServers: Record<string, {
    command: string;
    args: string[];
    env?: Record<string, string>;
  }>;
}

// MCP 工具目录条目（轻量级，用于按需加载）
interface ToolCatalogEntry {
  name: string;                    // 工具名，如 "filesystem:read_file"
  description: string;             // 简短描述（~10 tokens）
  mcpServer: string;              // 来源 MCP Server 名称
  inputSchema?: object;           // 完整 schema（load_tool 后才有）
}

// MCP 工具完整定义（load_tool 后返回）
interface ToolDefinition {
  name: string;
  description: string;
  inputSchema: {
    type: "object";
    properties: Record<string, any>;
    required?: string[];
  };
  mcpServer: string;
}

// MCP Server 运行时状态
interface McpServerStatus {
  name: string;
  status: "connected" | "disconnected" | "error" | "starting";
  pid?: number;                    // 子进程 PID
  toolCount?: number;              // 已注册工具数
  resourceCount?: number;          // 资源数
  promptCount?: number;            // 提示模板数
  lastError?: string;
  startedAt?: number;
  restartCount: number;
}

// MCP Resource
interface McpResource {
  uri: string;                     // 资源 URI，如 "file:///path/to/file"
  name: string;
  description?: string;
  mimeType?: string;
  mcpServer: string;
}

// MCP Prompt Template
interface McpPrompt {
  name: string;
  description: string;
  arguments?: { name: string; description: string; required?: boolean }[];
  mcpServer: string;
}
```

### 关键实现逻辑

**MCP Server 子进程管理与工具注册流程**：

```typescript
// main/services/mcp/McpClient.ts

import { Client } from "@modelcontextprotocol/sdk/client/index.js";
import { StdioClientTransport } from "@modelcontextprotocol/sdk/client/stdio.js";
import { spawn, ChildProcess } from "child_process";

class McpClientManager {
  private servers: Map<string, { client: Client; process: ChildProcess; status: McpServerStatus }> = new Map();
  private configPath = path.join(os.homedir(), ".agentbuddy/config/mcp.json");

  // 应用启动时调用
  async initialize(): Promise<void> {
    const config = await this.loadConfig();
    for (const [name, serverConfig] of Object.entries(config.mcpServers)) {
      const fullConfig: McpServerConfig = {
        name,
        command: serverConfig.command,
        args: serverConfig.args,
        env: serverConfig.env,
        permission: "confirm", // 默认权限
        enabled: true,
      };
      if (fullConfig.enabled) {
        await this.startServer(fullConfig);
      }
    }
  }

  async startServer(config: McpServerConfig): Promise<void> {
    // 1. 创建子进程
    const childProcess = spawn(config.command, config.args, {
      stdio: ["pipe", "pipe", "pipe"],
      env: { ...process.env, ...config.env },
    });

    // 2. 创建 MCP Client
    const transport = new StdioClientTransport(childProcess);
    const client = new Client(
      { name: "agent-buddy", version: "1.0.0" },
      { capabilities: {} }
    );

    await client.connect(transport);

    // 3. 调用 tools/list 发现工具
    const toolsResponse = await client.listTools();

    // 4. 为每个 MCP 工具注册轻量级目录（不注册完整 schema）
    for (const tool of toolsResponse.tools) {
      const toolName = `${config.name}:${tool.name}`;
      // 注册到 ContextBudgetManager 的工具目录
      this.registerToolCatalog(toolName, tool.description, config.name);
    }

    // 5. 注册按需加载的代理工具到 Pi
    // 注册一个代理工具，调用时再加载完整 schema 并转发到 MCP Server
    pi.registerTool({
      name: `mcp:${config.name}:*`, // 通配符代理
      description: `MCP Server ${config.name} tools`,
      parameters: { toolName: { type: "string" }, args: { type: "object" } },
      permissions: config.permission,
      execute: async (params) => {
        return await this.callMcpTool(config.name, params.toolName, params.args);
      },
    });

    // 6. 探测 resources 和 prompts
    try {
      const resources = await client.listResources();
      // 注册到 UI 资源浏览器
    } catch (e) { /* 部分 Server 不支持 resources */ }

    try {
      const prompts = await client.listPrompts();
      // 注册到 `/` 命令
    } catch (e) { /* 部分 Server 不支持 prompts */ }

    this.servers.set(config.name, { client, process: childProcess, status: { ... } });
  }

  // 按需加载工具完整定义
  async loadTool(toolName: string): Promise<ToolDefinition> {
    const [serverName, toolLocalName] = toolName.split(":");
    const server = this.servers.get(serverName);
    if (!server) throw new Error(`MCP Server ${serverName} not found`);

    const toolsResponse = await server.client.listTools();
    const tool = toolsResponse.tools.find(t => t.name === toolLocalName);
    if (!tool) throw new Error(`Tool ${toolLocalName} not found in ${serverName}`);

    return {
      name: toolName,
      description: tool.description,
      inputSchema: tool.inputSchema,
      mcpServer: serverName,
    };
  }

  // 转发工具调用到 MCP Server
  async callMcpTool(serverName: string, toolName: string, args: any): Promise<any> {
    const server = this.servers.get(serverName);
    if (!server) throw new Error(`MCP Server ${serverName} not running`);

    const result = await server.client.callTool({
      name: toolName,
      arguments: args,
    });

    return result.content;
  }

  // 应用关闭时清理
  async shutdown(): Promise<void> {
    for (const [name, { client, process }] of this.servers) {
      await client.close();
      process.kill("SIGTERM");
    }
    this.servers.clear();
  }

  // 子进程崩溃自动重启
  private setupProcessWatcher(config: McpServerConfig, childProcess: ChildProcess): void {
    childProcess.on("exit", (code) => {
      if (code !== 0 && this.shouldRestart(config.name)) {
        this.restartServer(config);
      }
    });
  }
}
```
